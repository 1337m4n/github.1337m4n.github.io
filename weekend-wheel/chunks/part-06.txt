nimationFrame(frame);
      return;
    }

    /* 轻微回弹落位：+1.6° -> 0°，增强机械阻尼感 */
    var settleStart=performance.now();
    var overshoot=1.6;
    function settle(ts){
      var st=Math.min(1,(ts-settleStart)/220);
      var back=overshoot*Math.sin(st*Math.PI)*(1-st);
      setRotorAngle(target+back);
      if(st<1){requestAnimationFrame(settle);return;}

      angle=normalize(target);
      setRotorAngle(angle);
      rotor.classList.remove("is-spinning");
      history.push(chosenIdx);
      resultEl.textContent=items[chosenIdx];

      if(noRepeatEl.checked){
        active=active.filter(function(idx){return idx!==chosenIdx;});
        angle=0;
        setRotorAngle(0);
        buildWheel();
      }

      renderHistory();
      spinning=false;
      updateRemain();
    }
    requestAnimationFrame(settle);
  }

  requestAnimationFrame(frame);
}
